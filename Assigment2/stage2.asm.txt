//Stage 2 – Single Player Input
//Match Stick Game
//Enter Name
      MOV R10, #NameInput
      STR R10, .WriteString 
      MOV R9, #Playername
      STR R9, .ReadString
      STR R9, .WriteString
//Enter MatchStick
prompt:
      MOV R10, #Question1
      STR R10, .WriteString
      LDR R0, .InputNum
      STR R0, .WriteUnsignedNum
      CMP R0, #10
      BLT prompt
      CMP R0, #100
      BGT prompt
//Print Output
loop:
      MOV R10, #NameDisplay
      STR R10, .WriteString
      STR R9, .WriteString
      MOV R10, #msg
      STR R10, . WriteString
      STR R0, .WriteUnsignedNum
      MOV R10, #matchsticknum
      STR R10, .WriteString
//Game Start
      MOV R10, #NameDisplay
      STR R10, .WriteString
      STR R9, .WriteString
      MOV R10, #Question2
      STR R10, .WriteString
      LDR R8, .InputNum
      CMP R8, #1 
      BLT loop
      CMP R8, #7
      BGT loop
      STR R8, .WriteUnsignedNum
//Calculation
      SUB R0,R0,R8
      CMP R0, #0 
      BGT loop 
done: 
      MOV R1, #finish
      STR R1, .WriteString
      HALT
//RequiredInput
NameInput: .ASCIZ "Please enter your name: "
Playername: .BLOCK 128
Question1: .ASCIZ "\nHow many matchsticks (10-100)?"
//Printout 
NameDisplay: .ASCIZ "\nPlayer "
msg:  .ASCIZ " ,there are "
matchsticknum: .ASCIZ "matchsticks remaining.\n"
Question2: .ASCIZ " , how many do you want to remove (1-7)?"
finish: .ASCIZ "\nGame Over\n"
